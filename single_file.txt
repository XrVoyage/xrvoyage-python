const { loadConfig, saveConfig } = require('./configHandler');

class ConfigManager {
    constructor() {
        this.config = null;
    }

    async loadConfig() {
        this.config = await loadConfig();
        return this.config;
    }

    saveConfig() {
        return saveConfig(this.config);
    }

    getConfig() {
        return this.config;
    }

    updateConfig(key, value) {
        if (this.config) {
            this.config[key] = value;
            this.saveConfig();
        }
    }
}

module.exports = ConfigManager;

### loadModulesList.js
const { loadConfig } = require('./configHandler');

async function loadModules() {
    try {
        const config = await loadConfig();
        if (!config || !Array.isArray(config.modules) || config.modules.length === 0) {
            throw new Error('Invalid or empty module configuration');
        }
        return config.modules;
    } catch (error) {
        console.error('Failed to load modules:', error);
        return [];
    }
}

module.exports = { loadModules };
### logManager.js
class LogManager {
    constructor() {
        this.logBuffer = {};
    }

    addLog(moduleName, logData) {
        if (!this.logBuffer[moduleName]) {
            this.logBuffer[moduleName] = '';
        }
        this.logBuffer[moduleName] += logData;
    }

    clearLog(moduleName) {
        this.logBuffer[moduleName] = '';
    }

    getLog(moduleName) {
        return this.logBuffer[moduleName] || '';
    }
}

module.exports = LogManager;

const { app, BrowserWindow, Menu, dialog, ipcMain } = require('electron');
const path = require('path');
const handleIPC = require('./src/ipcHandlers');
const { loadConfig, saveConfig } = require('./src/configHandler');
const log = require('./src/logger');

let mainWindow;

function createWindow() {
    log.info('Creating main window...');
    const config = loadConfig();
    const { width, height, devToolsOpen } = config.appSettings;

    app.commandLine.appendSwitch('disable-features', 'AutofillServerCommunication');

    mainWindow = new BrowserWindow({
        width,
        height,
        webPreferences: {
            nodeIntegration: true,
            contextIsolation: false,
        },
        icon: path.join(__dirname, 'assets', process.platform === 'win32' ? 'icon.ico' : 'icon.png')
    });

    mainWindow.loadFile('index.html');
    handleIPC(mainWindow);

    if (devToolsOpen) {
        mainWindow.webContents.openDevTools();
    }

    mainWindow.webContents.on('devtools-opened', () => {
        mainWindow.webContents.send('devtools-changed', true);
    });

    mainWindow.webContents.on('devtools-closed', () => {
        mainWindow.webContents.send('devtools-changed', false);
    });

    mainWindow.on('resize', () => {
        const [width, height] = mainWindow.getSize();
        config.appSettings.width = width;
        config.appSettings.height = height;
        saveConfig(config);
        log.info('Window size saved:', { width, height });
    });

    createMenu();

    log.info('Main window created');
}

function createMenu() {
    log.debug('Creating application menu');
    const config = loadConfig();
    const template = [
        {
            label: 'View',
            submenu: [
                {
                    label: 'Follow Logs',
                    type: 'checkbox',
                    checked: config.appSettings.followLogs,
                    click: () => {
                        config.appSettings.followLogs = !config.appSettings.followLogs;
                        saveConfig(config);
                        mainWindow.webContents.send('follow-logs-changed', config.appSettings.followLogs);
                        log.info('Follow logs setting changed:', config.appSettings.followLogs);
                    }
                }
            ]
        }
    ];
    const menu = Menu.buildFromTemplate(template);
    Menu.setApplicationMenu(menu);
    log.debug('Application menu created');
}

app.whenReady().then(() => {
    log.info('Electron app is ready');
    createWindow();
});

app.on('window-all-closed', () => {
    log.info('All windows closed');
    if (process.platform !== 'darwin') {
        app.quit();
    }
});

app.on('activate', () => {
    log.info('App activated');
    if (BrowserWindow.getAllWindows().length === 0) {
        createWindow();
    }
});

process.on('uncaughtException', (error) => {
    log.error('Uncaught exception:', error);
    app.quit();
});

ipcMain.on('show-error', (event, message) => {
    log.error('Error shown to user:', message);
    dialog.showErrorBox('Error', message);
});

log.info('Main process initialized');
### moduleList.js
const log = require('./logger');
const ModuleListItem = require('./moduleListItem');
const { loadModules } = require('./loadModulesList');

class ModuleList {
    constructor(containerId) {
        this.container = document.getElementById(containerId);
        if (!this.container) {
            log.error(`Container with id ${containerId} not found`);
            throw new Error(`Container with id ${containerId} not found`);
        }
        this.modules = [];
        log.debug('ModuleList constructor called');
    }

    async initialize() {
        log.debug('Initializing ModuleList');
        try {
            this.modules = await loadModules();
            log.info(`Loaded ${this.modules.length} modules`);
        } catch (error) {
            log.error('Failed to load modules:', error);
            this.modules = [];
        }
    }

    render() {
        log.debug('Rendering ModuleList');
        this.container.innerHTML = '';
        if (this.modules.length === 0) {
            log.warn('No modules available to render');
            this.renderNoModulesMessage();
        } else {
            this.modules.forEach((module, index) => {
                log.debug(`Creating ModuleListItem for ${module.name}`);
                const item = new ModuleListItem(module);
                this.container.appendChild(item.element);
                log.debug(`Added ModuleListItem ${index + 1}: ${module.name}`);
            });
        }
        log.info('ModuleList rendering complete');
    }

    renderNoModulesMessage() {
        log.debug('Rendering no modules message');
        const message = document.createElement('div');
        message.textContent = 'No modules available';
        message.className = 'no-modules-message';
        this.container.appendChild(message);
    }

    updateModuleStatus(moduleName, status) {
        log.debug(`Updating status for module: ${moduleName} to ${status}`);
        const module = this.modules.find(m => m.name === moduleName);
        if (module) {
            module.status = status;
            const item = this.container.querySelector(`.module-item[data-name="${moduleName}"]`);
            if (item) {
                const statusElement = item.querySelector('.module-status');
                statusElement.textContent = status;
                statusElement.className = `module-status ${status}`;
                log.info(`Module ${moduleName} status updated to ${status}`);
            } else {
                log.warn(`ModuleListItem for ${moduleName} not found in DOM`);
            }
        } else {
            log.error(`Module ${moduleName} not found in modules list`);
        }
    }
}

module.exports = ModuleList;

const log = require('./logger');

class ModuleListItem {
    constructor(module) {
        this.module = module;
        this.element = this.createElement();
        log.debug(`ModuleListItem created for ${module.name}`);
    }

    createElement() {
        log.debug(`Creating element for ModuleListItem: ${this.module.name}`);
        const element = document.createElement('div');
        element.className = 'module-item';
        element.setAttribute('data-name', this.module.name);
        element.innerHTML = `
            <div class="module-icon">${this.getModuleIcon()}</div>
            <span class="module-name">${this.module.name}</span>
            <span class="module-status">${this.module.status || 'stopped'}</span>
        `;
        element.addEventListener('click', () => this.onSelect());
        log.debug(`Element created for ModuleListItem: ${this.module.name}`);
        return element;
    }

    getModuleIcon() {
        log.debug(`Getting icon for module type: ${this.module.type}`);
        switch (this.module.type) {
            case 'python': return 'P';
            case 'docker': return 'D';
            case 'docker-compose': return 'C';
            default: 
                log.warn(`Unknown module type: ${this.module.type}`);
                return '?';
        }
    }

    onSelect() {
        log.info(`Module selected: ${this.module.name}`);
        // Implement selection logic here
    }

    updateStatus(status) {
        log.debug(`Updating status for ModuleListItem: ${this.module.name} to ${status}`);
        this.module.status = status;
        const statusElement = this.element.querySelector('.module-status');
        statusElement.textContent = status;
        statusElement.className = `module-status ${status}`;
        log.info(`Status updated for ModuleListItem: ${this.module.name} to ${status}`);
    }
}

module.exports = ModuleListItem;

import os
from logzero import logger, loglevel, INFO, DEBUG

loglevel(INFO)

def is_binary_file(filepath):
    binary_extensions = ('.jpg', '.jpeg', '.png', '.gif', '.ico', '.zip', '.tar', '.gz', '.rar', '.7z')
    return filepath.endswith(binary_extensions)

def should_skip_directory(directory):
    return 'node_modules' in directory

def process_directory(directory, output_file):
    for root, dirs, files in os.walk(directory):
        # Skip node_modules directory
        if should_skip_directory(root):
            logger.debug(f"Skipping directory: {root}")
            continue
        
        for file in files:
            filepath = os.path.join(root, file)
            # Skip binary files
            if is_binary_file(filepath):
                relative_path = os.path.relpath(filepath, directory)
                logger.debug(f"Skipping binary file: {relative_path}")
                continue

            # Write file content to output file
            with open(filepath, 'r', encoding='utf-8', errors='ignore') as f:
                relative_path = os.path.relpath(filepath, directory)
                output_file.write(f"\n### {relative_path}\n")
                output_file.write(f.read())
                logger.info(f"Added file: {relative_path}")

def main():

    with open(output_filename, 'w', encoding='utf-8') as output_file:
        process_directory(start_directory, output_file)

### src\ipcHandlers.js
const { ipcMain } = require('electron');
const { spawn, exec } = require('child_process');
const log = require('electron-log');
const path = require('path');
const os = require('os');
const { loadConfig, saveConfig } = require('./configHandler');

let moduleProcesses = {};
const config = loadConfig();
const modules = config.modules;

function handleIPC(mainWindow) {
    function checkContainerStatus(containerName) {
        return new Promise((resolve, reject) => {
            exec(`docker ps -q -f name=${containerName}`, (error, stdout, stderr) => {
                if (error) {
                    reject(error);
                    return;
                }
                resolve(stdout.trim() !== '');
            });
        });
    }

    async function startModule(module, event) {
        log.info(`Starting module: ${module.name}`);
        if (!moduleProcesses[module.name]) {
            event.sender.send('module-status-update', { moduleName: module.name, status: 'starting' });
            event.sender.send('clear-log', module.name);

            let moduleProcess;
            switch (module.type) {
                case 'python':
                    moduleProcess = spawn(
                        'conda',
                        ['run', '-n', 'xrvoyage', '--live-stream', `python ${module.path}`],
                        {
                            cwd: module.workingDir,
                            detached: false,
                            stdio: ['pipe', 'pipe', 'pipe'],
                            shell: true,
                            windowsHide: false,
                            env: { ...process.env, PYTHONUNBUFFERED: "1", FORCE_COLOR: "1" }
                        }
                    );
                    break;
                case 'docker':
                    const containerName = module.args ? module.args.find(arg => arg.startsWith('--name')) : null;
                    if (containerName) {
                        const actualName = containerName.split('=')[1];
                        const isRunning = await checkContainerStatus(actualName);
                        if (isRunning) {
                            log.info(`Container ${actualName} is already running. Attaching to logs.`);
                            moduleProcess = spawn('docker', ['logs', '-f', actualName], {
                                cwd: module.workingDir,
                                detached: false,
                                stdio: ['pipe', 'pipe', 'pipe'],
                                shell: false,
                                windowsHide: false
                            });
                        } else {
                            moduleProcess = spawn('docker', ['run', ...module.args, module.path], {
                                cwd: module.workingDir,
                                detached: false,
                                stdio: ['pipe', 'pipe', 'pipe'],
                                shell: false,
                                windowsHide: false
                            });
                        }
                    } else {
                        log.error(`No container name specified for module: ${module.name}`);
                        event.sender.send('module-status-update', { moduleName: module.name, status: 'stopped' });
                        return;
                    }
                    break;
                case 'docker-compose':
                    moduleProcess = spawn('docker-compose', ['up'], {
                        cwd: module.workingDir,
                        detached: false,
                        stdio: ['pipe', 'pipe', 'pipe'],
                        shell: false,
                        windowsHide: false
                    });
                    break;
            }

            function sendLogUpdate(data, isError = false) {
                const logData = data.toString();
                log.info(`Module ${isError ? 'stderr' : 'stdout'}: ${logData}`);
                mainWindow.webContents.send('log-update', { 
                    moduleName: module.name, 
                    logData: logData,
                    isError: isError
                });
            }

            moduleProcess.stdout.on('data', (data) => sendLogUpdate(data));
            moduleProcess.stderr.on('data', (data) => sendLogUpdate(data, true));

            moduleProcesses[module.name] = moduleProcess;

            log.info(`Module PID: ${moduleProcess.pid} for module ${module.name}`);
            event.sender.send('module-started', module.name);
            event.sender.send('module-status-update', { moduleName: module.name, status: 'running' });

            moduleProcess.on('exit', (code) => {
                log.info(`Module process ${moduleProcess.pid} exited with code ${code}`);
                delete moduleProcesses[module.name];
                event.sender.send('module-stopped', module.name);
                event.sender.send('module-status-update', { moduleName: module.name, status: 'stopped' });
            });
        } else {
            log.info(`Module ${module.name} is already running.`);
        }
    }

    ipcMain.on('start-module', (event, module) => {
        startModule(module, event);
    });

    ipcMain.on('stop-module', (event, module) => {
        log.info(`Stopping module: ${module.name}`);
        if (moduleProcesses[module.name]) {
            event.sender.send('module-status-update', { moduleName: module.name, status: 'stopping' });
            const process = moduleProcesses[module.name];
            try {
                process.kill('SIGTERM');
                log.info(`Module process ${process.pid} for module ${module.name} killed.`);
                delete moduleProcesses[module.name];
                event.sender.send('module-stopped', module.name);
                event.sender.send('module-status-update', { moduleName: module.name, status: 'stopped' });
            } catch (err) {
                log.error(`Failed to kill module process ${process.pid}: ${err}`);
                event.sender.send('module-status-update', { moduleName: module.name, status: 'unknown' });
            }
        }
    });

    async function checkModuleStatus(module) {
        if (module.type === 'docker' || module.type === 'docker-compose') {
            const containerName = module.type === 'docker' 
                ? (module.args ? module.args.find(arg => arg.startsWith('--name')) : null)
                : module.container_name;
            if (containerName) {
                const actualName = module.type === 'docker' ? containerName.split('=')[1] : containerName;
                try {
                    const isRunning = await checkContainerStatus(actualName);
                    return isRunning ? 'running' : 'stopped';
                } catch (error) {
                    log.error(`Error checking status for ${module.name}: ${error}`);
                    return 'unknown';
                }
            } else {
                log.error(`No container name specified for module: ${module.name}`);
                return 'unknown';
            }
        } else {
            return moduleProcesses[module.name] ? 'running' : 'stopped';
        }
    }

    setInterval(async () => {
        for (const module of modules) {
            const status = await checkModuleStatus(module);
            mainWindow.webContents.send('module-status-update', { moduleName: module.name, status });
        }
    }, 5000); // Check every 5 seconds

    // Handle application exit
    const shouldStopContainers = config.appSettings.stopContainersOnExit || false;
    if (shouldStopContainers) {
        process.on('exit', () => {
            modules.forEach(module => {
                if (module.type === 'docker' && module.args) {
                    const containerName = module.args.find(arg => arg.startsWith('--name'));
                    if (containerName) {
                        const actualName = containerName.split('=')[1];
                        exec(`docker stop ${actualName}`, (error) => {
                            if (error) {
                                log.error(`Failed to stop container ${actualName}: ${error}`);
                            } else {
                                log.info(`Stopped container ${actualName}`);
                            }
                        });
                    }
                } else if (module.type === 'docker-compose') {
                    exec('docker-compose down', { cwd: module.workingDir }, (error) => {
                        if (error) {
                            log.error(`Failed to stop docker-compose services: ${error}`);
                        } else {
                            log.info(`Stopped docker-compose services for ${module.name}`);
                        }
                    });
                }
            });
        });
    }
}

module.exports = handleIPC;

const winston = require('winston');
const path = require('path');

// ANSI color codes
const colors = {
  reset: "\x1b[0m",
  green: "\x1b[32m",
  lightBlue: "\x1b[94m",
  yellow: "\x1b[33m",
  red: "\x1b[31m",
  lightGray: "\x1b[37m"
};

const logger = winston.createLogger({
  level: 'debug',
  format: winston.format.combine(
    winston.format.timestamp({
      format: 'YYMMDD HH:mm:ss'
    }),
    winston.format.printf(({ level, message, timestamp, ...metadata }) => {
      const shortLevel = level.charAt(0).toUpperCase();
      let color;
      switch (level) {
        case 'info':
          color = colors.green;
          break;
        case 'debug':
          color = colors.lightBlue;
          break;
        case 'warn':
          color = colors.yellow;
          break;
        case 'error':
          color = colors.red;
          break;
        default:
          color = colors.reset;
      }
      return `${color}[${shortLevel} ${timestamp} ${metadata.file}:${metadata.line}]${colors.reset}${colors.lightGray} ${message}${colors.reset}`;
    })
  ),
  transports: [
    new winston.transports.Console(),
    new winston.transports.File({ 
      filename: 'app.log',
      format: winston.format.uncolorize()
    })
  ]
});

function getCallerInfo() {
  const err = new Error();
  const callerLine = err.stack.split('\n')[3];
  const match = callerLine.match(/at\s+(?:\w+\.)?(\w+)\s+\((.+):(\d+):\d+\)/);
  if (match) {
    return { 
      function: match[1], 
      file: path.basename(match[2]), 
      line: match[3] 
    };
  }
  return { function: 'unknown', file: 'unknown', line: '0' };
}

function createLogger() {
  return {
    debug: (message) => logger.debug(message, getCallerInfo()),
    info: (message) => logger.info(message, getCallerInfo()),
    warn: (message) => logger.warn(message, getCallerInfo()),
    error: (message) => logger.error(message, getCallerInfo())
  };
}

module.exports = createLogger();

### src\scriptManager.js
const log = require('./logger');

let cmdWindowState = {};

function toggleCMDWindow(pid) {
    try {
        const state = cmdWindowState[pid] || 'hidden';
        if (state === 'hidden') {
            log.info(`Would toggle window to visible for PID ${pid}`);
            cmdWindowState[pid] = 'visible';
        } else {
            log.info(`Would toggle window to hidden for PID ${pid}`);
            cmdWindowState[pid] = 'hidden';
        }
    } catch (err) {
        log.error(`Failed to toggle window for PID ${pid}: ${err}`);
    }
}

function clearCMDWindowState(pid) {
    log.info(`Clearing window state for PID ${pid}`);
    delete cmdWindowState[pid];
}

module.exports = { toggleCMDWindow, clearCMDWindowState };

### src\tray.js
const { Tray, Menu, app } = require('electron');
const path = require('path');
const log = require('./logger');
const { toggleCMDWindow } = require('./scriptManager');

function createTray(mainWindow, scriptProcesses) {
    const iconPath = path.join(__dirname, '../assets/icon.png');
    const tray = new Tray(iconPath);

    function updateContextMenu() {
        const contextMenu = Menu.buildFromTemplate([
            {
                label: 'Show App', click: function () {
                    mainWindow.show();
                    log.info('Main window restored from tray.');
                }
            },
            {
                label: 'Scripts',
                submenu: Object.keys(scriptProcesses).map(scriptPath => ({
                    label: path.basename(scriptPath),
                    click: () => {
                        const process = scriptProcesses[scriptPath];
                        if (process) {
                            const pid = process.pid;
                            toggleCMDWindow(pid);
                        }
                    }
                }))
            },
            {
                label: 'Quit', click: function () {
                    app.isQuiting = true;
                    app.quit();
                }
            }
        ]);

        tray.setContextMenu(contextMenu);
    }

    updateContextMenu();

    tray.on('click', function () {
        if (mainWindow.isVisible()) {
            mainWindow.hide();
            log.info('Main window hidden.');
        } else {
            mainWindow.show();
            log.info('Main window shown.');
        }
    });

    log.info('Tray initialized.');
    return tray;
}

module.exports = createTray;
